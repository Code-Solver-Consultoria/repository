1   package br.com.codesolver.scap.security;
2   
3   import java.lang.System.Logger;
4   import java.lang.System.Logger.Level;
5   import java.sql.Connection;
6   import java.sql.SQLException;
7   
8   import javax.naming.InitialContext;
9   import javax.naming.NamingException;
10  import javax.sql.DataSource;
11  
12  import edu.umd.cs.findbugs.annotations.SuppressFBWarnings;
13  
14  /**
15   * Gerenciamento de conexões com o banco de dados. 
16   *
17   * <p>Essa classe é uma implementação básica.</p>
18   *
19   * @author <a href="mailto:luciano@codesolver.com.br">Luciano Vieira Rodrigues</a>
20   * @since 2026-07-23
21   */
22  public class Database {
23  	
24  	/**
25  	 * Log da classe.
26  	 */
27  	private static final Logger LOGGER = System.getLogger(Database.class.getName());
28  	
29  	/**
30  	 * Propriedades para a configuração do gerenciador de conexões.
31  	 */
32  	private CSConfiguration configuration;
33  
34  	/**
35  	 * Acesso ao banco de dados pelo container.
36  	 */
37  	private DataSource dataSource;
38  	
39  	/**
40  	 * Construtor padrão.
41  	 *
42  	 * @param configuration Configuração para as conexões com banco de dados.
43  	 */
44  	@SuppressFBWarnings("CT_CONSTRUCTOR_THROW")
45  	public Database(CSConfiguration configuration) {
46  		this.configuration	= configuration;
47  		this.dataSource		= getDataSource();
48  	}
49  
50  	/**
51  	 * Recupera a conexão de banco de dados do container.
52  	 *
53  	 * @return DataSource gerenciado.
54  	 */
55  	private DataSource getDataSource() {
56  		DataSource result = null;
57  		InitialContext context;
58  		try {
59  			context = new InitialContext();
60  			result	= (DataSource) context.lookup(configuration.getDatasource()); 
61  		} catch (NamingException e) {
62  			String message = String.format("Erro ao recuperar a conexão do banco de dados \"%s\" do container.%n%s", 
63  					configuration.getDatasource(), e.getLocalizedMessage());
64  			LOGGER.log(Level.ERROR, message);
65  			throw new RuntimeException(message);
66  		}
67  		return result;
68  	}
69  
70  	/**
71  	 * Recupeera uma conexão disponível pelo DataSource.
72  	 *
73  	 * @return Conexão com o banco de dados.
74  	 * @throws SQLException
75  	 */
76  	public Connection getConnection() throws SQLException {
77  		return dataSource.getConnection();
78  	}
79  }
