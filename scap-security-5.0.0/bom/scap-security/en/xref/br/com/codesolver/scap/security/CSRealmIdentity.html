96      public CSRealmIdentity(Principal principal, CSConfiguration configuration) {
97          LOGGER.log(Level.TRACE, () -> String.format("Preparando a autenticação para o usuário \"%s\".", principal.getName()));
98          this.configuration	= configuration;
99          this.cache 			= CSPrincipalCache.getInstance();
100         if (cache.constains(principal.getName())) {
101             this.principal	= cache.get(principal.getName());	
102         } else {
103             this.principal	= new CSPrincipal(principal.getName());
104             this.cache.add(this.principal);
105         }
106         this.attributes		= new MapAttributes();
107         this.uuid			= new UUIDGenerator(configuration.getUUIDNode());
108     }

134     public SupportLevel getEvidenceVerifySupport(Class<? extends Evidence> evidenceType, String algorithmName)
135             throws RealmUnavailableException {
136         LOGGER.log(Level.TRACE, () -> String.format("Verificando o suporte ao tipo de evidência %s com o " +
137                     "algorítimo %s para o usupário %s.", 
138                     evidenceType, algorithmName, principal.getName()));

145     @Override
146     @SuppressWarnings({
147         "ReturnCount", "CyclomaticComplexity", "JavaNCSS", "NestedIfDepth",
148         "NPathComplexity", "ExecutableStatementCount", "MethodLength"
149     })
150     public boolean verifyEvidence(Evidence evidence) throws RealmUnavailableException {
151         LOGGER.log(Level.TRACE, () -> String.format("Validando a evidência de identificação do usuário %s", 
152                 principal.getName()));
153         if (evidence instanceof PasswordGuessEvidence) {
154             PasswordGuessEvidence password = (PasswordGuessEvidence) evidence;
155             Database database = new Database(configuration);
156             try (Connection connection = database.getConnection()) {
157                 Usuario usuario = findUsuario(connection, principal.getName());
158                 if (usuario == null) {
159                     LOGGER.log(Level.TRACE, () -> String.format(INVALID_USER, principal.getName()));
160                     return false;
161                 }
162                 if (!usuario.getAtivo()) {
163                     LOGGER.log(Level.WARNING, String.format("Usuário \"%s\" está inativo.", principal.getName()));
164                     return false;
165                 }
166                 // Compara as senhas:
167                 CharBuffer chars 	= CharBuffer.wrap(password.getGuess());
168                 ByteBuffer buffer 	= Charset.defaultCharset().encode(chars);
169                 byte[] bytes 		= Arrays.copyOfRange(buffer.array(), buffer.position(), buffer.limit());
170                 Digester digester	= DigesterFactory.getInstance(AlgorithmType.SHA_512);
171                 digester.update(bytes);
172                 char[] digested		= digester.digest().toCharArray();
173                 boolean valid       = Arrays.equals(digested, usuario.getSenha().toCharArray());;
174                 // Se a senha é válida e não está expirada:
175                 if (valid) {
176                     // Informações adicionais do usuaŕio:
177                     principal.setID(usuario.getId());
178                     principal.setFullName(usuario.getNome());
179                     principal.setExpired(false);
180                     principal.setOperator(usuario.getTipo() == UsuarioTipo.OPERADOR);
181                     cache.add(principal);
182                     // Arquivando as permissões:
183                     List<Funcionalidade> funcionalidades = findFuncionalidades(connection, usuario);
184                     funcionalidades.forEach(f -> processFuncionalidade(f));
185                     // Se o usuário for um operador de sistema, inicia a sessão de auditoria:
186                     if (principal.isOperator()) {
187                         Sessao sessao = new Sessao();
188                         sessao.setId(uuid.generate());
189                         sessao.setUsuario(usuario);
190                         sessao.setAbertura(LocalDateTime.now());
191                         sessao.setIp(getRemoteIP());
192                         registerSession(connection, sessao);
193                         principal.setSessionID(sessao.getId());
194                     }
195                 }
196                 return valid;
197             } catch (SQLException e) {
198                 LOGGER.log(Level.ERROR, String.format("Erro na camada de persistência: %s.", e.getMessage()));
199             }							// try (Database database ... (auto close)
200         }								// if (evidence instanceof PasswordGuessEvidence)
201         return false;
202     }

204     private void registerSession(Connection connection, Sessao sessao) throws SQLException {
205         String sql = """
206             insert into auditoria_sessao (
207                 id_auditoria_sessao,
208                 id_estacao,
209                 id_usuario,
210                 data_abertura,
211                 data_fechamento,
212                 ip_terminal)
213             values (
214                 ?,
215                 (select id_estacao from estacao where sigla = (select valor from parametro where id_parametro = 2)),
216                 ?,
217                 ?,
218                 NULL,
219                 ?)
220                 """;
221         try (PreparedStatement statement = connection.prepareStatement(sql)) {
222             statement.setObject(1, sessao.getId());
223             statement.setLong(2, sessao.getUsuario().getId());
224             statement.setObject(3, sessao.getAbertura());
225             statement.setString(4, sessao.getIp());
226             statement.executeUpdate();
227         }
228     }

230     private List<Funcionalidade> findFuncionalidades(Connection connection, Usuario usuario) throws SQLException {
231         String sql = """
232             select f.id_funcionalidade, f.nome, f.descricao, f.ativo
233               from funcionalidade f
234              inner join modulo m using (id_modulo)
235              inner join perfil_funcionalidade pf using (id_funcionalidade)
236              inner join perfil p using (id_perfil)
237              inner join usuario_perfil up using (id_perfil)
238              where id_usuario = ?
239                and m.ativo = true
240                and p.ativo = true
241              union
242             select f.id_funcionalidade, f.nome, f.descricao, f.ativo
243               from funcionalidade f
244              inner join modulo m using (id_modulo)
245              inner join usuario_funcionalidade uf using (id_funcionalidade)
246              where id_usuario = ?
247                and m.ativo = true
248                 """;
249         List<Funcionalidade> funcionalidades = new ArrayList<>();
250         try (PreparedStatement statement = connection.prepareStatement(sql)) {
251             statement.setLong(1, usuario.getId());
252             statement.setLong(2, usuario.getId());
253             try (ResultSet result = statement.executeQuery()) {
254                 while (result.next()) {
255                     Funcionalidade funcionalidade = new Funcionalidade();
256                     funcionalidade.setId(result.getString("id_funcionalidade"));
257                     funcionalidade.setNome(result.getString("nome"));
258                     funcionalidade.setDescricao(result.getString("descricao"));
259                     funcionalidade.setAtivo(result.getBoolean("ativo"));
260                     funcionalidades.add(funcionalidade);
261                 }
262             }
263         }
264         return funcionalidades;
265     }

267     /**
268      * Localiza um usuário por seu login.
269      *
270      * @param database Conexão com o banco de dados.
271      * @param name Nome de login do usuário.
272      * @return Usuario
273      * @throws SQLException Usuário não pode ser localizado.
274      */
275     private Usuario findUsuario(Connection connection, String name) throws SQLException {
276         String sql = """
277                 select id_usuario,
278                        id_usuario_tipo,
279                        nome,
280                        login,
281                        senha,
282                        matricula,
283                        cargo,
284                        setor,
285                        ativo,
286                        data_criacao
287                   from usuario
288                  where login = ?
289                 """;
290         try (PreparedStatement statement = connection.prepareStatement(sql)) {
291             statement.setString(1, name);
292             try (ResultSet result = statement.executeQuery()) {
293                 if (result.next()) {
294                     Usuario usuario = new Usuario();
295                     usuario.setId(result.getLong("id_usuario"));
296                     usuario.setTipo(UsuarioTipo.valueOf(result.getLong("id_usuario_tipo")));
297                     usuario.setNome(result.getString("nome"));
298                     usuario.setLogin(result.getString("login"));
299                     usuario.setSenha(result.getString("senha"));
300                     usuario.setMatricula(result.getString("matricula"));
301                     usuario.setCargo(result.getString("cargo"));
302                     usuario.setSetor(result.getString("setor"));
303                     usuario.setAtivo(result.getBoolean("ativo"));
304                     usuario.setData(result.getObject("data_criacao", LocalDateTime.class));
305                     return usuario;
306                 }
307             }
308         }
309         return null;
310     }

312     /**
313      * Adiciona as regras de acesso do usuário.
314      *
315      * @param funcionalidade Funcionalidade de autorização do sistema.
316      */
317     private void processFuncionalidade(Funcionalidade funcionalidade) {
318         if (funcionalidade.getAtivo()) {
319             if (attributes.containsKey(ROLES_IDENTITY)) {
320                 attributes.get(ROLES_IDENTITY).add(funcionalidade.getId());
321             } else {
322                 attributes.addFirst(ROLES_IDENTITY, funcionalidade.getId());
323             }
324         }
325     }

339     /**
340      * Obtém o endereço IP do usuário remoto.
341      *
342      * <p>
343      * Este método so consegue obter o endereçp do usuário quando a conexão remote
344      * for através de http.
345      *
346      * @return retorna o IP do usuário remoto
347      */
348     private String getRemoteIP() {
349         String result = null;
350         try {
351             ServletRequest request = (HttpServletRequest) PolicyContext.getContext("jakarta.servlet.http.HttpServletRequest");
352             if (request != null) {
353                 result = request.getRemoteAddr();
354             }
355         } catch (IllegalArgumentException | SecurityException | PolicyContextException e) {
356             LOGGER.log(Level.WARNING, 
357                 String.format("Não foi possivel obter o IP do usuário \"%s\" remoto para registrar na auditoria",
358                     principal.getName()));
359         }
360         return result;
361     }
